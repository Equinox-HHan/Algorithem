# 找到字符串中所有字母异位词

## 题目：
给定两个字符串s和p找到s中所有p的异位词的子串，返回这些子串的起始索引不考虑答案输出的顺序

## 解题思路
异位词问题配合定长滑动窗口
先覆盖基础长度然后不需要每次重新进行词频统计只需要更新首尾词频

## 代码
- CPP
```
vector<int> func(string s,string p)
{
    if(s.size()<p.size())
    {
        return {};//防御性编程
    }
    int k=p.size();
    vector<int> ans;
    vector<int> isExist(26,0);
    vector<int> count(26,0);
    for(char c:p)
    {
        isExist[c-'a']+=1;
    }
    for(int i=0;i<k;i+=1)
    {
        count[s[i]-'a']+=1;
    }
    if(count==isExist)
    {
        ans.push_back{0};
    }
    for(int pos=1;pos<=s.size()-k;pos+=1)
    {
        count[s[pos-1]-'a']-=1;
        count[s[pos+k-1]-'a']+=1;
        if(count==isExist)
        {
            ans.push_back(pos);
        }
    }
    return ans;
}
```

def func(s,p):
    if len(s)<len(p):
        return []
    ans,isExist,count=[],[0]*26,[0]*26
    k=len(p)
    for i in range(k):
        isExist[p[i]-ord('a')]+=1
        count[s[i]-ord('a')]+=1
    if count==isExist:
        ans.append(0)
    for pos in range(1,len(s)-k+1):
        count[s[pos-1]-ord('a')]-=1
        count[s[pos+k-1]-ord('a')]+=1
        if count==isExist:
            ans.append(pos)
    return ans
```

```
def checkInclusion(s1: str, s2: str) -> bool:
    if len(s2) < len(s1):
        return False
    
    isExist = [0] * 26
    count = [0] * 26
    k = len(s1)
    
    # 初始化第一个窗口
    for i in range(k):
        count[ord(s2[i]) - ord('a')] += 1
        isExist[ord(s1[i]) - ord('a')] += 1
        
    if count == isExist:
        return True
        
    # 注意 Python 的 range 顾头不顾尾，上限应为 len(s2) - k + 1
    for pos in range(1, len(s2) - k + 1):
        count[ord(s2[pos - 1]) - ord('a')] -= 1
        count[ord(s2[pos + k - 1]) - ord('a')] += 1
        if count == isExist:
            return True
            
    return False
