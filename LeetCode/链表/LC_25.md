# k组一个反转链表

## 题目：
给你链表的头节点 head ，每 k 个节点一组进行翻转，请你返回修改后的链表
k 是一个正整数，它的值小于或等于链表的长度。如果节点总数不是 k 的整数倍，那么请将最后剩余的节点保持原有顺序
你不能只是单纯的改变节点内部的值，而是需要实际进行节点交换

## 题目分析：
这是一道模板类型的题目：重点就是在反转的时候找到关键节点

## 代码：
- CPP
```
class Solution {
public:
    ListNode* reverseKGroup(ListNode* head, int k) {
        ListNode dummy(0);
        ListNode *kPre=&dummy;
        kPre->next=head;

        while(true){
            ListNode *kEnd=kPre;
            for(int i=0;i<k&&kEnd!=nullptr;i++){
                kEnd=kEnd->next;
            }
            if(kEnd==nullptr){
                break;
            }
            ListNode *group_start=kPre->next;
            ListNode *new_group_start=kEnd->next;
            ListNode *newHead=reverseNode(group_start,new_group_start);
            kPre->next=newHead;
            group_start->next=new_group_start;
            kPre=group_start;
        }
        return dummy.next;
    }
    ListNode *reverseNode(ListNode *head,ListNode *tail){
        ListNode *pre=nullptr;
        ListNode *cur=head;
        while(cur!=tail){
            ListNode *temp=cur->next;
            cur->next=pre;
            pre=cur;
            cur=temp;
        }
        return pre;
    }
};
```
- PY
```
# Definition for singly-linked list.
class ListNode:
    def __init__(self, val=0, next=None):
        self.val = val
        self.next = next

class Solution:
    def reverseKGroup(self, head: ListNode, k: int) -> ListNode:
        dummy = ListNode(0)
        kPre = dummy
        kPre.next = head

        while True:
            kEnd = kPre
            for i in range(k):
                if kEnd is None:
                    break
                kEnd = kEnd.next
            
            if kEnd is None:
                break
            
            group_start = kPre.next
            new_group_start = kEnd.next
            newHead = self.reverseNode(group_start, new_group_start)
            
            kPre.next = newHead
            group_start.next = new_group_start
            kPre = group_start

        return dummy.next

    def reverseNode(self, head: ListNode, tail: ListNode) -> ListNode:
        pre = None
        cur = head
        while cur != tail:
            temp = cur.next
            cur.next = pre
            pre = cur
            cur = temp
        return pre
```
