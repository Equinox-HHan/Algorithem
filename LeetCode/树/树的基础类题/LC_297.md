# 二叉树的序列化和反序列化

## 题目
序列化是将一个数据结构或者对象转换为连续的比特位的操作，进而可以将转换后的数据存储在一个文件或者内存中，
同时也可以通过网络传输到另一个计算机环境，采取相反方式重构得到原数据

## 分析：
二叉树的序列化和反序列化主要有前序和层序两种方法
前者主要思想为DFS，后者主要思想为BFS
(前序的反序列化注意参数一定要引用传参，否则每层递归的状态会被打乱)

## 代码：
- CPP
```
#include<bits/stdc++.h>
using namespace std;
//前序序列化
class Codec {
public:

    // Encodes a tree to a single string.
    string serialize(TreeNode* root) {
        if(!root){
            return "#,";
        }
        return to_string(root->val)+","+serialize(root->left)+serialize(root->right);
    }

    // Decodes your encoded data to tree.
    TreeNode* deserialize(string data) {
        stringstream ss(data);
        string node;
        queue<string> nodes;
        while(getline(ss,node,',')){
            if(!node.empty()){
                nodes.push(node);
            }
        }
        return help_func(nodes);
    }
    
private:
    TreeNode* help_func(queue<string> &nodes){
        if(nodes.empty()){
            return nullptr;
        }
        string node=nodes.front();
        nodes.pop();
        if(node=="#"){
            return nullptr;
        }
        TreeNode* root=new TreeNode(stoi(node));
        root->left=help_func(nodes);
        root->right=help_func(nodes);
        return root;
    }
};
```

```
#include<bits/stdc++.h>
using namespace std;

//层序序列化和反序列化
class Codec {
public:

    // Encodes a tree to a single string.
    string serialize(TreeNode* root) {
        if(!root){
            return "";
        }
        queue<TreeNode*> level;
        level.push(root);
        string ans="";
        
        while(!level.empty()){
            TreeNode* node=level.front();
            level.pop();
            if(node){
                string value=to_string(node->val);
                ans=ans+value+",";
                level.push(node->left);
                level.push(node->right);
            }
            else{
                ans+="#,";
            }
        }
        return ans;
    }

    // Decodes your encoded data to tree.
    TreeNode* deserialize(string data) {
        if(data.empty()){
            return nullptr;
        }
        stringstream ss(data);
        string node;
        queue<TreeNode*> nodes;

        getline(ss,node,',');
        TreeNode* head=new TreeNode(stoi(node));
        nodes.push(head);

        while(!nodes.empty()){
            TreeNode* parent=nodes.front();
            nodes.pop();
            if(getline(ss,node,',')){
                if(node!="#"){
                    parent->left=new TreeNode(stoi(node));
                    nodes.push(parent->left);
                }
            }
            if(getline(ss,node,',')){
                if(node!="#"){
                    parent->right=new TreeNode(stoi(node));
                    nodes.push(parent->right);
                }
            }
        }
        return head;
    }
};
```
