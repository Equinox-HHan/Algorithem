# 螺旋矩阵

## 题目：
给你一个 m 行 n 列的矩阵 matrix 请按照顺时针螺旋顺序返回矩阵中的所有元素

## 题目分析：
四个方向设置四堵墙做夹逼，每次遍历完行列就直接收缩墙
要做全局穿透判断：只要出现穿透就终止大循环返回答案
left，right，top，buttom都是闭区间，表示可以覆盖到的最外层

## 代码：
- CPP
```
class Solution {
public:
    vector<int> spiralOrder(vector<vector<int>>& matrix) {
        vector<int> ans;
        if (matrix.empty()) return ans;

        // 定义四面“墙”
        int top = 0;
        int bottom = matrix.size() - 1;
        int left = 0;
        int right = matrix[0].size() - 1;

        while (true) {
            // 1. 向右走：遍历 top 行，从 left 到 right
            for (int i = left; i <= right; ++i) ans.push_back(matrix[top][i]);
            if (++top > bottom) break; // 墙往内缩，如果穿透了，立即退出！

            // 2. 向下走：遍历 right 列，从 top 到 bottom
            for (int i = top; i <= bottom; ++i) ans.push_back(matrix[i][right]);
            if (--right < left) break;

            // 3. 向左走：遍历 bottom 行，从 right 到 left
            for (int i = right; i >= left; --i) ans.push_back(matrix[bottom][i]);
            if (--bottom < top) break;

            // 4. 向上走：遍历 left 列，从 bottom 到 top
            for (int i = bottom; i >= top; --i) ans.push_back(matrix[i][left]);
            if (++left > right) break;
        }

        return ans;
    }
};
```
- PY
```
class Solution:

    def spiralOrder(self, matrix: list[list[int]]) -> list[int]:
        if not matrix:
            return []

        ans = []
        top = 0
        bottom = len(matrix) - 1
        left = 0
        right = len(matrix[0]) - 1

        while True:
            # 1. 向右走：遍历 top 行，从 left 到 right
            for i in range(left, right + 1):
                ans.append(matrix[top][i])
            top += 1  # 上墙下移
            if top > bottom:  # 穿透检测
                break

            # 2. 向下走：遍历 right 列，从 top 到 bottom
            for i in range(top, bottom + 1):
                ans.append(matrix[i][right])
            right -= 1  # 右墙左移
            if right < left:
                break

            # 3. 向左走：遍历 bottom 行，从 right 到 left
            for i in range(right, left - 1, -1):
                ans.append(matrix[bottom][i])
            bottom -= 1  # 下墙上移
            if bottom < top:
                break

            # 4. 向上走：遍历 left 列，从 bottom 到 top
            for i in range(bottom, top - 1, -1):
                ans.append(matrix[i][left])
            left += 1  # 左墙右移
            if left > right:
                break

        return ans
```
